                      Rosa #FF3366 | Branco quente #FAFAFA
   Entrega: 2 relatorios
     A) [BS2 Dash Consolidado HTML]   — visao executiva consolidada
     B) [BS2 Dash Sinais Vitais HTML] — painel operacional de atendentes
   Criar TODAS as medidas na tabela fato_genesys, NA ORDEM ABAIXO.

/* =========================================================================
   BLOCO 1 — MEDIDAS BASE (volume, TMA, comparativos)
   ========================================================================= */

Atendimentos Realizados =
VAR _Qtd = DISTINCTCOUNT ( fato_genesys[conversation_id] )
RETURN IF ( ISBLANK ( _Qtd ), 0, _Qtd )


Atendentes Ativos =
VAR _Qtd =
    CALCULATE (
        DISTINCTCOUNT ( fato_genesys[participant_name] ),
        KEEPFILTERS ( NOT ISBLANK ( fato_genesys[participant_name] ) )
    )
RETURN IF ( ISBLANK ( _Qtd ), 0, _Qtd )


Media Atendimentos por Atendente =
DIVIDE ( [Atendimentos Realizados], [Atendentes Ativos] )


TMA Segundos =
AVERAGEX (
    VALUES ( fato_genesys[conversation_id] ),
    VAR _Ini = CALCULATE ( MIN ( fato_genesys[conversation_start] ) )
    VAR _Fim = CALCULATE ( MAX ( fato_genesys[conversation_end] ) )
    RETURN
        IF (
            NOT ISBLANK ( _Ini ) && NOT ISBLANK ( _Fim ) && _Fim >= _Ini,
            DATEDIFF ( _Ini, _Fim, SECOND )
        )
)


TMA Formatado =
VAR _Seg = [TMA Segundos]
VAR _H = INT ( DIVIDE ( _Seg, 3600 ) )
VAR _M = INT ( DIVIDE ( MOD ( _Seg, 3600 ), 60 ) )
VAR _S = INT ( MOD ( _Seg, 60 ) )
RETURN
    IF (
        ISBLANK ( _Seg ),
        "--:--:--",
        FORMAT ( _H, "00" ) & ":" & FORMAT ( _M, "00" ) & ":" & FORMAT ( _S, "00" )
    )


/* TMA medio do portfolio — ignora o filtro de atendente para servir de referencia
   na classificacao de status (PREMISSA: nao ha meta de TMA documentada).        */
TMA Portfolio Segundos =
CALCULATE ( [TMA Segundos], REMOVEFILTERS ( fato_genesys[participant_name] ) )


Periodo Analisado =
VAR _Ini = MIN ( fato_genesys[Data] )
VAR _Fim = MAX ( fato_genesys[Data] )
RETURN
    IF (
        ISBLANK ( _Ini ),
        "Sem periodo",
        FORMAT ( _Ini, "DD/MM/YYYY" ) & " a " & FORMAT ( _Fim, "DD/MM/YYYY" )
    )


/* --- Comparativo mensal (PREMISSA: sem Dim_Calendario; usa fato_genesys[Data]) --- */

Atendimentos Mes Atual =
VAR _U = MAX ( fato_genesys[Data] )
VAR _I = EOMONTH ( _U, -1 ) + 1
VAR _F = EOMONTH ( _U, 0 )
RETURN
    CALCULATE (
        [Atendimentos Realizados],
        REMOVEFILTERS ( fato_genesys[Data] ),
        fato_genesys[Data] >= _I,
        fato_genesys[Data] <= _F
    )


Atendimentos Mes Anterior =
VAR _U = MAX ( fato_genesys[Data] )
VAR _I = EOMONTH ( _U, -2 ) + 1
VAR _F = EOMONTH ( _U, -1 )
RETURN
    CALCULATE (
        [Atendimentos Realizados],
        REMOVEFILTERS ( fato_genesys[Data] ),
        fato_genesys[Data] >= _I,
        fato_genesys[Data] <= _F
    )


Var % Atendimentos MoM =
VAR _A = [Atendimentos Mes Atual]
VAR _B = [Atendimentos Mes Anterior]
RETURN IF ( _B = 0 || ISBLANK ( _B ), BLANK (), DIVIDE ( _A - _B, _B ) )


TMA Mes Atual Segundos =
VAR _U = MAX ( fato_genesys[Data] )
VAR _I = EOMONTH ( _U, -1 ) + 1
VAR _F = EOMONTH ( _U, 0 )
RETURN
    CALCULATE (
        [TMA Segundos],
        REMOVEFILTERS ( fato_genesys[Data] ),
        fato_genesys[Data] >= _I,
        fato_genesys[Data] <= _F
    )


TMA Mes Anterior Segundos =
VAR _U = MAX ( fato_genesys[Data] )
VAR _I = EOMONTH ( _U, -2 ) + 1
VAR _F = EOMONTH ( _U, -1 )
RETURN
    CALCULATE (
        [TMA Segundos],
        REMOVEFILTERS ( fato_genesys[Data] ),
        fato_genesys[Data] >= _I,
        fato_genesys[Data] <= _F
    )


Var % TMA MoM =
VAR _A = [TMA Mes Atual Segundos]
VAR _B = [TMA Mes Anterior Segundos]
RETURN IF ( _B = 0 || ISBLANK ( _B ), BLANK (), DIVIDE ( _A - _B, _B ) )


/* --- Top canal / top departamento --- */

Canal Mais Atendido =
VAR _T = ADDCOLUMNS ( VALUES ( fato_genesys[media_type] ), "@Q", [Atendimentos Realizados] )
VAR _N = MAXX ( TOPN ( 1, _T, [@Q], DESC, fato_genesys[media_type], ASC ), fato_genesys[media_type] )
RETURN IF ( ISBLANK ( _N ), "Sem dados", _N )


Qtd Canal Mais Atendido =
VAR _T = ADDCOLUMNS ( VALUES ( fato_genesys[media_type] ), "@Q", [Atendimentos Realizados] )
VAR _V = MAXX ( TOPN ( 1, _T, [@Q], DESC, fato_genesys[media_type], ASC ), [@Q] )
RETURN IF ( ISBLANK ( _V ), 0, _V )


% Canal Mais Atendido = DIVIDE ( [Qtd Canal Mais Atendido], [Atendimentos Realizados] )


Departamento Mais Acionado =
VAR _T = ADDCOLUMNS ( VALUES ( fato_genesys[department] ), "@Q", [Atendimentos Realizados] )
VAR _N = MAXX ( TOPN ( 1, _T, [@Q], DESC, fato_genesys[department], ASC ), fato_genesys[department] )
RETURN IF ( ISBLANK ( _N ), "Sem dados", _N )


Qtd Departamento Mais Acionado =
VAR _T = ADDCOLUMNS ( VALUES ( fato_genesys[department] ), "@Q", [Atendimentos Realizados] )
VAR _V = MAXX ( TOPN ( 1, _T, [@Q], DESC, fato_genesys[department], ASC ), [@Q] )
RETURN IF ( ISBLANK ( _V ), 0, _V )


% Departamento Mais Acionado = DIVIDE ( [Qtd Departamento Mais Acionado], [Atendimentos Realizados] )


/* --- Semaforo de atendentes (PREMISSA: limiares relativos a media do portfolio) ---
   NORMAL    : TMA do atendente <  media do portfolio
   MONITORAR : media <= TMA < 1,5 x media
   CRITICO   : TMA >= 1,5 x media                                                  */

Atendentes em Alerta =
VAR _Ref = [TMA Portfolio Segundos]
VAR _T =
    ADDCOLUMNS (
        CALCULATETABLE (
            VALUES ( fato_genesys[participant_name] ),
            KEEPFILTERS ( NOT ISBLANK ( fato_genesys[participant_name] ) )
        ),
        "@TMA", [TMA Segundos]
    )
VAR _Q = COUNTROWS ( FILTER ( _T, [@TMA] >= _Ref && [@TMA] < _Ref * 1.5 ) )
RETURN IF ( ISBLANK ( _Q ), 0, _Q )


Atendentes Criticos =
VAR _Ref = [TMA Portfolio Segundos]
VAR _T =
    ADDCOLUMNS (
        CALCULATETABLE (
            VALUES ( fato_genesys[participant_name] ),
            KEEPFILTERS ( NOT ISBLANK ( fato_genesys[participant_name] ) )
        ),
        "@TMA", [TMA Segundos]
    )
VAR _Q = COUNTROWS ( FILTER ( _T, [@TMA] >= _Ref * 1.5 ) )
RETURN IF ( ISBLANK ( _Q ), 0, _Q )

/* =========================================================================
   BLOCO 3 — RELATORIO A: [BS2 Dash Consolidado HTML]
   Visao executiva consolidada — KPIs, linha mensal, rankings, insights
   ========================================================================= */

BS2 Dash Consolidado HTML =
VAR Q = UNICHAR ( 34 )
VAR _Total = [Atendimentos Realizados]

/* --- KPIs --- */
VAR _TotalFmt   = FORMAT ( _Total, "#,##0" )
VAR _TMA        = [TMA Formatado]
VAR _Periodo    = [Periodo Analisado]
VAR _Atendentes = FORMAT ( [Atendentes Ativos], "#,##0" )
VAR _MediaAt    = FORMAT ( [Media Atendimentos por Atendente], "#,##0.0" )
VAR _CanalTop   = [Canal Mais Atendido]
VAR _CanalPct   = FORMAT ( [% Canal Mais Atendido], "0.0%" )
VAR _CanalQtd   = FORMAT ( [Qtd Canal Mais Atendido], "#,##0" )
VAR _DepTop     = [Departamento Mais Acionado]
VAR _DepPct     = FORMAT ( [% Departamento Mais Acionado], "0.0%" )
VAR _DepQtd     = FORMAT ( [Qtd Departamento Mais Acionado], "#,##0" )

VAR _VarVol   = [Var % Atendimentos MoM]
VAR _PillVol =
    IF (
        ISBLANK ( _VarVol ),
        "<span class=" & Q & "pill neu" & Q & ">&#9679; s/ base</span> comparativa",
        "<span class=" & Q & "pill " & IF ( _VarVol >= 0, "pos", "neg" ) & Q & ">"
            & IF ( _VarVol >= 0, "&#9650; ", "&#9660; " ) & FORMAT ( ABS ( _VarVol ), "0.0%" )
            & "</span> vs mes anterior"
    )

/* TMA: queda e resultado positivo -> classe invertida */
VAR _VarTMA = [Var % TMA MoM]
VAR _PillTMA =
    IF (
        ISBLANK ( _VarTMA ),
        "<span class=" & Q & "pill neu" & Q & ">&#9679; s/ base</span> comparativa",
        "<span class=" & Q & "pill " & IF ( _VarTMA <= 0, "pos", "neg" ) & Q & ">"
            & IF ( _VarTMA <= 0, "&#9660; ", "&#9650; " ) & FORMAT ( ABS ( _VarTMA ), "0.0%" )
            & "</span> vs mes anterior"
    )

/* --- Ranking generico: canal / departamento / fila --- */
VAR _TabCanal = ADDCOLUMNS ( VALUES ( fato_genesys[media_type] ), "@Q", [Atendimentos Realizados] )
VAR _TopCanal = TOPN ( 5, _TabCanal, [@Q], DESC, fato_genesys[media_type], ASC )
VAR _MaxCanal = MAXX ( _TopCanal, [@Q] )
VAR _LinCanal =
    CONCATENATEX (
        ADDCOLUMNS ( _TopCanal, "@P", RANKX ( _TopCanal, [@Q],, DESC, DENSE ) ),
        VAR _Nome = IF ( ISBLANK ( fato_genesys[media_type] ), "(nao informado)", fato_genesys[media_type] )
        VAR _W = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _MaxCanal ) * 100, "0.0" ), ",", "." )
        VAR _S = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _Total ) * 100, "0.0" ), ",", "." )
        RETURN
            "<tr><td class=" & Q & "p" & Q & ">" & FORMAT ( [@P], "0" ) & "</td>"
                & "<td class=" & Q & "l" & Q & ">" & _Nome & "</td>"
                & "<td><div class=" & Q & "bg" & Q & "><div class=" & Q & "br" & Q
                & " style=" & Q & "width:" & _W & "%" & Q & "></div></div></td>"
                & "<td class=" & Q & "q" & Q & ">" & FORMAT ( [@Q], "#,##0" ) & "</td>"
                & "<td class=" & Q & "c" & Q & ">" & _S & "%</td></tr>",
        "", [@Q], DESC
    )

VAR _TabDep = ADDCOLUMNS ( VALUES ( fato_genesys[department] ), "@Q", [Atendimentos Realizados] )
VAR _TopDep = TOPN ( 5, _TabDep, [@Q], DESC, fato_genesys[department], ASC )
VAR _MaxDep = MAXX ( _TopDep, [@Q] )
VAR _LinDep =
    CONCATENATEX (
        ADDCOLUMNS ( _TopDep, "@P", RANKX ( _TopDep, [@Q],, DESC, DENSE ) ),
        VAR _Nome = IF ( ISBLANK ( fato_genesys[department] ), "(nao informado)", fato_genesys[department] )
        VAR _W = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _MaxDep ) * 100, "0.0" ), ",", "." )
        VAR _S = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _Total ) * 100, "0.0" ), ",", "." )
        RETURN
            "<tr><td class=" & Q & "p" & Q & ">" & FORMAT ( [@P], "0" ) & "</td>"
                & "<td class=" & Q & "l" & Q & ">" & _Nome & "</td>"
                & "<td><div class=" & Q & "bg" & Q & "><div class=" & Q & "br" & Q
                & " style=" & Q & "width:" & _W & "%" & Q & "></div></div></td>"
                & "<td class=" & Q & "q" & Q & ">" & FORMAT ( [@Q], "#,##0" ) & "</td>"
                & "<td class=" & Q & "c" & Q & ">" & _S & "%</td></tr>",
        "", [@Q], DESC
    )

VAR _TabFila = ADDCOLUMNS ( VALUES ( fato_genesys[name_queue] ), "@Q", [Atendimentos Realizados] )
VAR _TopFila = TOPN ( 5, _TabFila, [@Q], DESC, fato_genesys[name_queue], ASC )
VAR _MaxFila = MAXX ( _TopFila, [@Q] )
VAR _LinFila =
    CONCATENATEX (
        ADDCOLUMNS ( _TopFila, "@P", RANKX ( _TopFila, [@Q],, DESC, DENSE ) ),
        VAR _Nome = IF ( ISBLANK ( fato_genesys[name_queue] ), "(nao informado)", fato_genesys[name_queue] )
        VAR _W = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _MaxFila ) * 100, "0.0" ), ",", "." )
        VAR _S = SUBSTITUTE ( FORMAT ( DIVIDE ( [@Q], _Total ) * 100, "0.0" ), ",", "." )
        RETURN
            "<tr><td class=" & Q & "p" & Q & ">" & FORMAT ( [@P], "0" ) & "</td>"
                & "<td class=" & Q & "l" & Q & ">" & _Nome & "</td>"
                & "<td><div class=" & Q & "bg" & Q & "><div class=" & Q & "br2" & Q
                & " style=" & Q & "width:" & _W & "%" & Q & "></div></div></td>"
                & "<td class=" & Q & "q" & Q & ">" & FORMAT ( [@Q], "#,##0" ) & "</td>"
                & "<td class=" & Q & "c" & Q & ">" & _S & "%</td></tr>",
        "", [@Q], DESC
    )

/* --- Serie mensal: volume (linha) + TMA (colunas) --- */
VAR _Meses = SUMMARIZE ( ADDCOLUMNS ( SUMMARIZE ( ALLSELECTED ( fato_genesys ), fato_genesys[Data] ),
                "@M", EOMONTH ( fato_genesys[Data], 0 ) ), [@M] )
VAR _MesesCalc =
    ADDCOLUMNS (
        _Meses,
        "@Vol",
            VAR _Mes = [@M]
            RETURN
                CALCULATE ( [Atendimentos Realizados], ALLSELECTED ( fato_genesys ),
                    FILTER ( ALL ( fato_genesys[Data] ), EOMONTH ( fato_genesys[Data], 0 ) = _Mes ) ),
        "@Tma",
            VAR _Mes2 = [@M]
            RETURN
                CALCULATE ( [TMA Segundos], ALLSELECTED ( fato_genesys ),
                    FILTER ( ALL ( fato_genesys[Data] ), EOMONTH ( fato_genesys[Data], 0 ) = _Mes2 ) )
    )
VAR _Top12 = TOPN ( 12, _MesesCalc, [@M], DESC )
VAR _Serie = ADDCOLUMNS ( _Top12, "@P", RANKX ( _Top12, [@M],, ASC, DENSE ) )
VAR _N = COUNTROWS ( _Serie )
VAR _MaxVol = MAXX ( _Serie, [@Vol] )
VAR _MaxTma = MAXX ( _Serie, [@Tma] )
VAR _X0 = 54
VAR _X1 = 872
VAR _YT = 30
VAR _YB = 152
VAR _Passo = IF ( _N > 1, DIVIDE ( _X1 - _X0, _N - 1 ), 0 )

VAR _Pontos =
    CONCATENATEX (
        _Serie,
        VAR _X = _X0 + ( [@P] - 1 ) * _Passo
        VAR _Y = _YB - DIVIDE ( [@Vol], _MaxVol ) * ( _YB - _YT )
        RETURN
            SUBSTITUTE ( FORMAT ( _X, "0.0" ), ",", "." ) & ","
                & SUBSTITUTE ( FORMAT ( _Y, "0.0" ), ",", "." ),
        " ", [@M], ASC
    )

VAR _Marcas =
    CONCATENATEX (
        _Serie,
        VAR _X = _X0 + ( [@P] - 1 ) * _Passo
        VAR _Y = _YB - DIVIDE ( [@Vol], _MaxVol ) * ( _YB - _YT )
        VAR _Xs = SUBSTITUTE ( FORMAT ( _X, "0.0" ), ",", "." )
        VAR _Ys = SUBSTITUTE ( FORMAT ( _Y, "0.0" ), ",", "." )
        VAR _Yt = SUBSTITUTE ( FORMAT ( _Y - 12, "0.0" ), ",", "." )
        RETURN
            "<circle cx=" & Q & _Xs & Q & " cy=" & Q & _Ys & Q & " r=" & Q & "3.2" & Q
                & " fill=" & Q & "#1226AA" & Q & " stroke=" & Q & "#FFFFFF" & Q
                & " stroke-width=" & Q & "1.4" & Q & "></circle>"
                & "<text x=" & Q & _Xs & Q & " y=" & Q & _Yt & Q & " font-size=" & Q & "12.5" & Q
                & " fill=" & Q & "#090E29" & Q & " text-anchor=" & Q & "middle" & Q
                & " font-weight=" & Q & "700" & Q & ">" & FORMAT ( [@Vol], "#,##0" ) & "</text>"
                & "<text x=" & Q & _Xs & Q & " y=" & Q & "186" & Q & " font-size=" & Q & "12.5" & Q
                & " fill=" & Q & "#6B7280" & Q & " text-anchor=" & Q & "middle" & Q & ">"
                & FORMAT ( [@M], "MMM/YY" ) & "</text>",
        "", [@M], ASC
    )

VAR _GradeVol =
    CONCATENATEX (
